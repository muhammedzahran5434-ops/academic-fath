   351→   351→            <button className="absolute right-5 top-5 flex h-11 w-11 items-center justify-center rounded-full bg-white/15 text-xl text-white backdrop-blur hover:bg-white/25" onClick={() => setPreview(null)}>✕</button>
   352→   352→          </motion.div>
   353→   353→        )}
   354→   354→      </AnimatePresence>
   355→   355→
   356→   356→      <Confirm open={!!delFile} onClose={() => setDelFile(null)} onConfirm={() => delFile && remove(delFile)} message={`Delete "${delFile?.name}" from the server permanently?`} />
   357→   357→    </div>
   358→   358→  )
   359→   359→}

   362→   362→export function Academic() {

   371→   371→  const load = useCallback(async () => {

   396→   396→      toast('Marks saved ✓', 'success')
   397→   397→      setEditing({})
   398→   398→      await load()
   399→   399→    } catch (e) {

   408→   408→      await api('/api/subjects', { method: 'POST', body: JSON.stringify({ action: 'submit-marks', subjectId, marks: val }) })
   409→   409→      toast('Marks submitted for approval 📨', 'success')
   410→   410→      setMyPending({})
   411→   411→      await load()
   412→   412→    } catch (e) {

   422→   422→      setNewSubj({ name: '', max: '50' })
   423→   423→      await load()
   424→   424→    } catch (e) {
   425→   425→      toast(e instanceof Error ? e.message : 'Failed', 'error')
   426→   426→    }
   427→   427→  }
   428→   428→
   429→   429→  const removeSubject = async (s: Subject) => {
   430→   430→    try {
   431→   431→      await api('/api/subjects', { method: 'POST', body: JSON.stringify({ action: 'remove-subject', subjectId: s.id }) })
   432→   432→      toast('Subject removed', 'success')
   433→   433→      await load()
   434→   434→    } catch (e) {
   435→   435→      toast(e instanceof Error ? e.message : 'Failed', 'error')
   436→   436→    }
   437→   437→  }
   438→   438→
   439→   439→  const approvePending = async (markId: string) => {

   441→   441→      await api('/api/subjects', { method: 'POST', body: JSON.stringify({ action: 'approve-marks', markId }) })
   442→   442→      toast('Marks approved ✓', 'success')
   443→   443→      await load()
   444→   444→    } catch (e) {
   445→   445→      toast(e instanceof Error ? e.message : 'Failed', 'error')
